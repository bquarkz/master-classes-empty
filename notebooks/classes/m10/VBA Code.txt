Sub GetStockData()

    Dim http As Object
    Dim JSON As Object
    Dim i As Integer
    Dim ws As Worksheet
    
    Set ws = ThisWorkbook.Sheets(1)
    ws.Cells.Clear

    ' Create request
    Set http = CreateObject("MSXML2.ServerXMLHTTP")
    
    http.Open "GET", "https://your-app-name.onrender.com/us-stocks", False
    http.send

    ' Parse JSON
    Set JSON = JsonConverter.ParseJson(http.responseText)

    ' Headers
    ws.Cells(1, 1).Value = "Ticker"
    ws.Cells(1, 2).Value = "Max Value"
    ws.Cells(1, 3).Value = "Min Value"
    ws.Cells(1, 4).Value = "Current Value"
    ws.Cells(1, 5).Value = "Avg Return %"
    ws.Cells(1, 6).Value = "Volatility %"

    ' Fill data
    For i = 1 To JSON.Count
        ws.Cells(i + 1, 1).Value = JSON(i)("Ticker")
        ws.Cells(i + 1, 2).Value = JSON(i)("Max_Value")
        ws.Cells(i + 1, 3).Value = JSON(i)("Min_Value")
        ws.Cells(i + 1, 4).Value = JSON(i)("Current_Value")
        ws.Cells(i + 1, 5).Value = JSON(i)("Avg_Daily_Return_%")
        ws.Cells(i + 1, 6).Value = JSON(i)("Volatility_%")
    Next i

    MsgBox "Stock data updated!"

End Sub
